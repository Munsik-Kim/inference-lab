"""Version routing algorithm
Purpose: Select a supported protocol version for a client and a server.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Choose the highest common supported version. If the intersection is empty, return an explicit incompatibility result rather than assuming a legacy protocol works.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
def negotiate(client_versions, server_versions, disabled=()):
    client = set(client_versions)
    server = set(server_versions)
    blocked = set(disabled)
    if any(not isinstance(v, int) or v < 1 for v in client | server):
        raise ValueError('invalid protocol version')
    common = sorted((client & server) - blocked)
    if not common:
        return {'status': 'incompatible', 'version': None}
    return {'status': 'selected', 'version': common[-1]}

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
