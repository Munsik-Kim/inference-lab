"""Tree traversal algorithm
Purpose: Walk a directory-like tree while preventing accidental cycles.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Mark a node before visiting its children. Encountering an already active node signals a cycle; encountering a completed shared node can reuse its recorded summary.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
def postorder(graph, root):
    active, complete, result = set(), set(), []
    def visit(node):
        if node in active:
            raise ValueError('cycle')
        if node in complete:
            return
        active.add(node)
        for child in graph.get(node, []):
            visit(child)
        active.remove(node)
        complete.add(node)
        result.append(node)
    visit(root)
    return result

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
