"""Ledger reconciliation algorithm
Purpose: Pair invoices and payments without hiding duplicate identifiers.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Match on identifier and account before comparing integer cents. Keep unmatched rows on both sides and reject duplicate keys rather than choosing the last value.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
def reconcile(invoices, payments):
    def index(rows):
        result = {}
        for row in rows:
            key = (row['record_id'], row['account_id'])
            if key in result:
                raise ValueError('duplicate key')
            result[key] = row['amount_cents']
        return result
    left, right = index(invoices), index(payments)
    return [{'key': key, 'invoice': left.get(key), 'payment': right.get(key),
             'matched': key in left and key in right and left[key] == right[key]}
            for key in sorted(set(left) | set(right))]

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
