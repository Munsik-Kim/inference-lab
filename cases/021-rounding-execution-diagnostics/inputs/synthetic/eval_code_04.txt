"""Configuration validation algorithm
Purpose: Validate a small configuration before opening any service resources.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Collect every field error in a stable order. An unknown field is reported separately from a missing required field so that a misspelling cannot silently select a default.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
def validate_config(config, limits):
    errors = []
    for name in sorted(set(config) - set(limits)):
        errors.append((name, 'unknown'))
    for name, (lower, upper) in sorted(limits.items()):
        if name not in config:
            errors.append((name, 'missing'))
            continue
        value = config[name]
        if type(value) is not int:
            errors.append((name, 'not integer'))
        elif not lower <= value <= upper:
            errors.append((name, 'outside inclusive limits'))
    return errors

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
