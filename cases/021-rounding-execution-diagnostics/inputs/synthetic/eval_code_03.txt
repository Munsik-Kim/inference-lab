"""Sliding counter algorithm
Purpose: Maintain a bounded time window for monitoring a queue.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Discard events strictly older than the left boundary and retain events exactly on it. Out of order timestamps must be rejected before the accumulator is changed.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
from collections import deque
class WindowCounter:
    def __init__(self, width):
        if width <= 0:
            raise ValueError('nonpositive width')
        self.width, self.events, self.total = width, deque(), 0
        self.last_time = None
    def add(self, timestamp, count):
        if count < 0 or (self.last_time is not None and timestamp < self.last_time):
            raise ValueError('invalid event order or count')
        self.last_time = timestamp
        while self.events and self.events[0][0] < timestamp - self.width:
            self.total -= self.events.popleft()[1]
        self.events.append((timestamp, count))
        self.total += count
        return self.total

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
