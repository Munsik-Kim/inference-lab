"""Inventory reservation algorithm
Purpose: Implement deterministic allocation without creating stock that does not exist.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Allocate at most the nonnegative usable stock. Preserve request order for equal priorities and record shortages instead of silently removing incomplete orders.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
def reserve(stock, requests):
    remaining = dict(stock)
    decisions = []
    ordered = sorted(enumerate(requests), key=lambda row: (-row[1]['priority'], row[0]))
    for original_index, request in ordered:
        item = request['item']
        amount = request['quantity']
        if amount < 0:
            raise ValueError('negative request')
        granted = min(amount, max(0, remaining.get(item, 0)))
        remaining[item] = remaining.get(item, 0) - granted
        decisions.append((original_index, item, granted, amount - granted))
    return sorted(decisions), remaining

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
