"""Stream frame parser
Purpose: Decode length-prefixed messages from arbitrary input chunks.
This is an invented example with no real people, facilities, or measurements. The numeric values are illustrative, not operating advice.
Operating rule: Do not consume a partial header or payload. A length above the configured cap is an error before allocation; successful frames advance the cursor exactly once.

Implementation review notes: Validate input before changing state. Preserve a stable ordering and return explicit errors. The examples below are synthetic fixtures, not production defaults.
"""
class FrameParser:
    def __init__(self, cap):
        self.cap, self.buffer = cap, bytearray()
    def feed(self, chunk):
        self.buffer.extend(chunk)
        frames, cursor = [], 0
        while len(self.buffer) - cursor >= 4:
            length = int.from_bytes(self.buffer[cursor:cursor+4], 'big')
            if length > self.cap:
                raise ValueError('oversized frame')
            if len(self.buffer) - cursor < 4 + length:
                break
            frames.append(bytes(self.buffer[cursor+4:cursor+4+length]))
            cursor += 4 + length
        del self.buffer[:cursor]
        return frames

from dataclasses import dataclass

@dataclass(frozen=True)
class Observation:
    entity: str
    values: dict

def validate_record(record):
    errors = []
    if not record.entity:
        errors.append("entity is required")
    for name, value in sorted(record.values.items()):
        if not isinstance(value, int) or value < 0:
            errors.append((name, "expected nonnegative integer"))
    return errors

def review_records(records):
    reviewed = []
    for sequence, record in enumerate(records):
        errors = validate_record(record)
        reviewed.append({"sequence": sequence, "entity": record.entity,
                         "errors": errors, "accepted_for_review": not errors})
    return reviewed
